def style_run(run, size=14, bold=False):
    run.font.name = 'Times New Roman'
    run._element.rPr.rFonts.set(qn('w:eastAsia'), 'Times New Roman')
    run.font.size = Pt(size)
    run.bold = bold

def add_p(text, size=14, bold=False, align=WD_ALIGN_PARAGRAPH.LEFT, after=12):
    p = doc.add_paragraph()
    p.alignment = align
    p.paragraph_format.line_spacing = 1.5
    p.paragraph_format.space_after = Pt(after)
    style_run(p.add_run(text), size, bold)

def add_h(text):
    p = doc.add_heading(text, level=1)
    p.alignment = WD_ALIGN_PARAGRAPH.LEFT

        style_run(run, 14, True)
        run.font.color.rgb = RGBColor(0, 0, 0)
    p.paragraph_format.space_before, p.paragraph_format.space_after = Pt(24), Pt(12)

add_h("3.3. Физическое проектирование (SQL DDL)")
sql_ddl = """CREATE TABLE Patient (patient_id INT PRIMARY KEY, full_name VARCHAR(100) NOT NULL, gender CHAR(1) CHECK (gender IN ('М', 'Ж')), birth_date DATE NOT NULL, phone VARCHAR(20), address TEXT);
CREATE TABLE Doctor (doctor_id INT PRIMARY KEY, full_name VARCHAR(100) NOT NULL, specialization VARCHAR(50));
CREATE TABLE Examination (exam_id INT PRIMARY KEY, patient_id INT REFERENCES Patient(patient_id), doctor_id INT REFERENCES Doctor(doctor_id), exam_date DATE NOT NULL, location VARCHAR(50) CHECK (location IN ('Прием', 'На дому')), symptoms TEXT NOT NULL, diagnosis TEXT, prescriptions TEXT);
CREATE TABLE Medicine (medicine_id INT PRIMARY KEY, name VARCHAR(100) NOT NULL, administration_method VARCHAR(50), action_description TEXT, side_effects TEXT);
CREATE TABLE Prescription_Link (prescription_id INT PRIMARY KEY, exam_id INT REFERENCES Examination(exam_id), medicine_id INT REFERENCES Medicine(medicine_id), dosage VARCHAR(50));"""

add_h("3.4. Разработка SQL-запросов (Задания 1-5)")
sql_q = """-- 1) Пациенты с >= 3 сходными симптомами
SELECT p.full_name, e.diagnosis, e.symptoms FROM Patient p JOIN Examination e ON p.patient_id = e.patient_id WHERE (e.symptoms LIKE '%температура%' AND e.symptoms LIKE '%кашель%' AND e.symptoms LIKE '%слабость%');
-- 2) Пациенты в диапазон дат у каждого врача
SELECT d.full_name AS doctor, p.full_name AS patient, e.exam_date FROM Doctor d JOIN Examination e ON d.doctor_id = e.doctor_id JOIN Patient p ON e.patient_id = p.patient_id WHERE e.exam_date BETWEEN '2026-01-01' AND '2026-10-31' ORDER BY d.full_name;
-- 3) Количество пациентов на дому у каждого врача
SELECT d.full_name, COUNT(DISTINCT e.patient_id) AS home_count FROM Doctor d JOIN Examination e ON d.doctor_id = e.doctor_id WHERE e.location = 'На дому' GROUP BY d.doctor_id, d.full_name;
-- 4) Распределение больных по возрасту в текущем году
SELECT CASE WHEN (EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM p.birth_date)) < 18 THEN '0-17' WHEN (EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM p.birth_date)) <= 35 THEN '18-35' WHEN (EXTRACT(YEAR FROM CURRENT_DATE) - EXTRACT(YEAR FROM p.birth_date)) <= 60 THEN '36-60' ELSE '60+' END AS age, COUNT(DISTINCT p.patient_id) FROM Patient p JOIN Examination e ON p.patient_id = e.patient_id WHERE EXTRACT(YEAR FROM e.exam_date) = EXTRACT(YEAR FROM CURRENT_DATE) GROUP BY age;
-- 5) Диагнозы по двум симптомам
SELECT DISTINCT diagnosis FROM Examination WHERE symptoms LIKE '%температура%' AND symptoms LIKE '%головная боль%';"""
p = doc.add_paragraph()
p.paragraph_format.line_spacing = 1.5
run = p.add_run(sql_q)
run.font.name, run.font.size = 'Courier New', Pt(10)
